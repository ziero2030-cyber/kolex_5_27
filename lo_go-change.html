document.addEventListener("DOMContentLoaded", () => { const logos = document.querySelectorAll(".logo-wrap .logo-img"); let currentIndex = 0; setInterval(() => { // 현재 활성화된 로고에서 active 클래스 제거 logos[currentIndex].classList.remove("active"); // 다음 로고 인덱스로 전환 (0 -> 1 -> 0 -> 1) currentIndex = (currentIndex + 1) % logos.length; // 새로운 로고에 active 클래스 추가 logos[currentIndex].classList.add("active"); }, 3000); // 3000ms = 3초 });
